1. Servidor Web Básico

IPTables

#!/bin/bash
# Configuración básica para servidor web

# Limpiar reglas existentes
iptables -F
iptables -X
iptables -Z

# Políticas por defecto
iptables -P INPUT DROP
iptables -P OUTPUT ACCEPT
iptables -P FORWARD DROP

# Permitir loopback
iptables -A INPUT -i lo -j ACCEPT

# Permitir conexiones establecidas
iptables -A INPUT -m state --state ESTABLISHED,RELATED -j ACCEPT

# HTTP/HTTPS
iptables -A INPUT -p tcp --dport 80 -j ACCEPT
iptables -A INPUT -p tcp --dport 443 -j ACCEPT

# SSH con límite de conexiones
iptables -A INPUT -p tcp --dport 22 -m state --state NEW \
    -m recent --set --name SSH
iptables -A INPUT -p tcp --dport 22 -m state --state NEW \
    -m recent --update --seconds 60 --hitcount 4 --rttl \
    --name SSH -j DROP

# Guardar reglas
netfilter-persistent save

UFW

#!/bin/bash
# Configuración UFW para servidor web

# Reset y configuración inicial
ufw reset
ufw default deny incoming
ufw default allow outgoing

# Permitir SSH con límite
ufw limit ssh

# Permitir HTTP/HTTPS
ufw allow 'Apache Full'
# O alternativamente:
ufw allow 80/tcp
ufw allow 443/tcp

# Activar firewall
ufw --force enable

2. Servidor de Base de Datos

MySQL/MariaDB

IPTables

# Permitir MySQL solo desde red local
iptables -A INPUT -p tcp -s 192.168.1.0/24 --dport 3306 -j ACCEPT

# Limitar conexiones por segundo
iptables -A INPUT -p tcp --dport 3306 -m limit \
    --limit 5/second --limit-burst 10 -j ACCEPT

UFW

# Permitir MySQL desde red local
ufw allow from 192.168.1.0/24 to any port 3306

# Permitir acceso desde IP específica
ufw allow from 192.168.1.100 to any port 3306

6. Configuraciones de Alta Seguridad

Protección contra Ataques Comunes

# Protección contra SYN Flood
iptables -A INPUT -p tcp --syn -m limit --limit 1/s -j ACCEPT

# Protección contra Port Scanning
iptables -A INPUT -p tcp --tcp-flags ALL NONE -j DROP
iptables -A INPUT -p tcp --tcp-flags ALL ALL -j DROP

# Anti-spoofing
iptables -A INPUT -s 127.0.0.0/8 ! -i lo -j DROP
iptables -A INPUT -s 0.0.0.0/8 -j DROP
iptables -A INPUT -s 169.254.0.0/16 -j DROP
iptables -A INPUT -s 172.16.0.0/12 -j DROP
iptables -A INPUT -s 192.168.0.0/16 -j DROP
iptables -A INPUT -s 224.0.0.0/4 -j DROP
iptables -A INPUT -d 224.0.0.0/4 -j DROP
iptables -A INPUT -s 240.0.0.0/5 -j DROP

# Bloquear ping
iptables -A INPUT -p icmp --icmp-type echo-request -j DROP

8. Scripts de Automatización

Script de Configuración Completa

#!/bin/bash
# Script completo de configuración de firewall

# Función para verificar privilegios de root
check_root() {
    if [ "$(id -u)" != "0" ]; then
        echo "Este script debe ejecutarse como root" 1>&2
        exit 1
    fi
}

# Función para backup
backup_rules() {
    mkdir -p /root/firewall_backup
    date_backup=$(date +%Y%m%d_%H%M%S)
    iptables-save > /root/firewall_backup/iptables_$date_backup.backup
}

# Función principal
setup_firewall() {
    # Limpiar reglas existentes
    iptables -F
    iptables -X
    iptables -Z

    # Políticas por defecto
    iptables -P INPUT DROP
    iptables -P OUTPUT ACCEPT
    iptables -P FORWARD DROP

    # Reglas básicas
    iptables -A INPUT -i lo -j ACCEPT
    iptables -A INPUT -m state --state ESTABLISHED,RELATED -j ACCEPT

    # SSH con protección
    iptables -A INPUT -p tcp --dport 22 -m state --state NEW \
        -m recent --set --name SSH
    iptables -A INPUT -p tcp --dport 22 -m state --state NEW \
        -m recent --update --seconds 60 --hitcount 4 --rttl \
        --name SSH -j DROP

    # HTTP/HTTPS
    iptables -A INPUT -p tcp --dport 80 -j ACCEPT
    iptables -A INPUT -p tcp --dport 443 -j ACCEPT

    # Protección adicional
    iptables -A INPUT -p tcp --tcp-flags ALL NONE -j DROP
    iptables -A INPUT -p tcp --tcp-flags ALL ALL -j DROP
    iptables -A INPUT -f -j DROP

    # Guardar reglas
    if command -v netfilter-persistent &> /dev/null; then
        netfilter-persistent save
    else
        iptables-save > /etc/iptables/rules.v4
    fi
}

# Ejecución principal
check_root
backup_rules
setup_firewall

9. Monitoreo y Logging

Configuración de Logs

# IPTables con logging
iptables -A INPUT -j LOG --log-prefix "IPTables-Dropped: " \
    --log-level 4

# UFW logging
ufw logging on
ufw logging high

⚠️ Consideraciones de Seguridad

  • Siempre hacer backup antes de aplicar cambios
  • Probar las reglas en un entorno de desarrollo
  • Mantener una sesión de emergencia activa
  • Documentar todos los cambios
  • Realizar pruebas de penetración después de los cambios

✅ Mejores Prácticas

  • Usar el principio de menor privilegio
  • Implementar logging y monitoreo
  • Actualizar regularmente las reglas
  • Mantener copias de seguridad
  • Documentar todas las configuraciones
