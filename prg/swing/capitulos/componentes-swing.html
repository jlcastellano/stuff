Componentes Swing en Java

1. Componentes de Texto

JTextField

Campo de texto de una sola línea.

JTextField textField = new JTextField(20); // 20 columnas
textField.setText("Texto inicial");
textField.setEditable(true);

JTextArea

Campo de texto multilínea.

JTextArea textArea = new JTextArea(5, 20); // 5 filas, 20 columnas
textArea.setLineWrap(true);
textArea.setWrapStyleWord(true);

JPasswordField

Campo para contraseñas.

JPasswordField passwordField = new JPasswordField(20);
passwordField.setEchoChar('*');

JEditorPane

Editor de texto con formato.

JEditorPane editorPane = new JEditorPane();
editorPane.setContentType("text/html");
editorPane.setText("Texto con formato");

2. Componentes de Selección

JButton

Botón clásico.

JButton button = new JButton("Clic aquí");
button.addActionListener(e -> System.out.println("Botón presionado"));

JToggleButton

Botón con dos estados.

JToggleButton toggleButton = new JToggleButton("Toggle");
toggleButton.addActionListener(e -> {
    boolean selected = toggleButton.isSelected();
    System.out.println("Estado: " + selected);
});

JCheckBox

Casilla de verificación.

JCheckBox checkBox = new JCheckBox("Activar opción");
checkBox.setSelected(true);

JRadioButton

Botón de radio para selección única.

ButtonGroup group = new ButtonGroup();
JRadioButton radio1 = new JRadioButton("Opción 1");
JRadioButton radio2 = new JRadioButton("Opción 2");
group.add(radio1);
group.add(radio2);

3. Componentes de Listado

JComboBox

Lista desplegable.

String[] opciones = {"Opción 1", "Opción 2", "Opción 3"};
JComboBox comboBox = new JComboBox<>(opciones);

JList

Lista de selección.

String[] elementos = {"Item 1", "Item 2", "Item 3"};
JList list = new JList<>(elementos);
list.setSelectionMode(ListSelectionModel.SINGLE_SELECTION);

JSpinner

Selector numérico.

JSpinner spinner = new JSpinner(new SpinnerNumberModel(0, 0, 100, 1));
spinner.addChangeListener(e -> System.out.println("Valor: " + spinner.getValue()));

4. Componentes de Diálogo

JFileChooser

Selector de archivos.

JFileChooser fileChooser = new JFileChooser();
fileChooser.setFileSelectionMode(JFileChooser.FILES_AND_DIRECTORIES);
int result = fileChooser.showOpenDialog(null);
if (result == JFileChooser.APPROVE_OPTION) {
    File selectedFile = fileChooser.getSelectedFile();
}

JColorChooser

Selector de colores.

Color color = JColorChooser.showDialog(null, "Seleccione un color", Color.WHITE);

JOptionPane

Diálogos de mensaje y entrada.

// Mensaje de información
JOptionPane.showMessageDialog(null, "Información", "Título", JOptionPane.INFORMATION_MESSAGE);

// Diálogo de confirmación
int option = JOptionPane.showConfirmDialog(null, "¿Está seguro?");

// Diálogo de entrada
String input = JOptionPane.showInputDialog("Ingrese un valor:");

5. Componentes de Visualización

JLabel

Etiqueta de texto o imagen.

JLabel label = new JLabel("Texto");
label.setIcon(new ImageIcon("imagen.png"));

JProgressBar

Barra de progreso.

JProgressBar progressBar = new JProgressBar(0, 100);
progressBar.setValue(50);
progressBar.setStringPainted(true);

JSlider

Control deslizante.

JSlider slider = new JSlider(JSlider.HORIZONTAL, 0, 100, 50);
slider.setMajorTickSpacing(20);
slider.setMinorTickSpacing(5);
slider.setPaintTicks(true);
slider.setPaintLabels(true);

6. Componentes de Tabla

JTable

Tabla de datos.

String[] columnNames = {"Nombre", "Edad", "Ciudad"};
Object[][] data = {
    {"Juan", 25, "Madrid"},
    {"Ana", 30, "Barcelona"}
};
JTable table = new JTable(data, columnNames);
JScrollPane scrollPane = new JScrollPane(table);

JTree

Árbol jerárquico.

DefaultMutableTreeNode root = new DefaultMutableTreeNode("Root");
DefaultMutableTreeNode child = new DefaultMutableTreeNode("Child");
root.add(child);
JTree tree = new JTree(root);

7. Componentes de Menú

JMenuBar

Barra de menú.

JMenuBar menuBar = new JMenuBar();
JMenu fileMenu = new JMenu("Archivo");
JMenuItem openItem = new JMenuItem("Abrir");
fileMenu.add(openItem);
menuBar.add(fileMenu);

JPopupMenu

Menú contextual.

JPopupMenu popup = new JPopupMenu();
JMenuItem cutItem = new JMenuItem("Cortar");
popup.add(cutItem);

// Mostrar en evento de clic derecho
component.addMouseListener(new MouseAdapter() {
    public void mouseReleased(MouseEvent e) {
        if (e.isPopupTrigger()) {
            popup.show(e.getComponent(), e.getX(), e.getY());
        }
    }
});

Ejemplo Completo de Aplicación

import javax.swing.*;
import java.awt.*;

public class EjemploComponentes extends JFrame {
    public EjemploComponentes() {
        setTitle("Demo de Componentes Swing");
        setSize(800, 600);
        setDefaultCloseOperation(JFrame.EXIT_ON_CLOSE);
        
        // Panel principal
        JPanel mainPanel = new JPanel(new BorderLayout(10, 10));
        mainPanel.setBorder(BorderFactory.createEmptyBorder(10, 10, 10, 10));
        
        // Barra de menú
        JMenuBar menuBar = new JMenuBar();
        JMenu fileMenu = new JMenu("Archivo");
        fileMenu.add(new JMenuItem("Nuevo"));
        fileMenu.add(new JMenuItem("Abrir"));
        fileMenu.add(new JMenuItem("Guardar"));
        menuBar.add(fileMenu);
        setJMenuBar(menuBar);
        
        // Panel izquierdo
        JPanel leftPanel = new JPanel(new GridLayout(0, 1, 5, 5));
        leftPanel.add(new JLabel("Nombre:"));
        leftPanel.add(new JTextField(20));
        leftPanel.add(new JLabel("Contraseña:"));
        leftPanel.add(new JPasswordField(20));
        leftPanel.add(new JCheckBox("Recordar datos"));
        leftPanel.add(new JButton("Enviar"));
        
        // Panel derecho
        JPanel rightPanel = new JPanel(new BorderLayout(5, 5));
        String[] columnNames = {"Nombre", "Edad"};
        Object[][] data = {{"Juan", 25}, {"Ana", 30}};
        JTable table = new JTable(data, columnNames);
        rightPanel.add(new JScrollPane(table), BorderLayout.CENTER);
        
        // Panel inferior
        JPanel bottomPanel = new JPanel(new FlowLayout());
        bottomPanel.add(new JProgressBar(0, 100));
        bottomPanel.add(new JButton("Cancelar"));
        
        // Agregar todo al panel principal
        mainPanel.add(leftPanel, BorderLayout.WEST);
        mainPanel.add(rightPanel, BorderLayout.CENTER);
        mainPanel.add(bottomPanel, BorderLayout.SOUTH);
        
        setContentPane(mainPanel);
    }
    
    public static void main(String[] args) {
        SwingUtilities.invokeLater(() -> {
            new EjemploComponentes().setVisible(true);
        });
    }
}

Mejores Prácticas

  • Usar SwingUtilities.invokeLater() para crear y modificar componentes
  • Implementar listeners para manejar eventos
  • Utilizar layouts apropiados
  • Mantener una estructura organizada de componentes
  • Validar entrada de usuario
  • Proporcionar retroalimentación visual al usuario
  • Considerar la accesibilidad
