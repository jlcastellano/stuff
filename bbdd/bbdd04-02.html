4.3. Lenguaje de Definición de Datos (DDL)

4.3.1. Definición de Bases de Datos

Crear Base de Datos

CREATE DATABASE nombre_bd
    [CHARACTER SET charset]
    [COLLATE collation];

-- Ejemplo
CREATE DATABASE empresa
    CHARACTER SET utf8mb4
    COLLATE utf8mb4_unicode_ci;

Modificar Base de Datos

ALTER DATABASE nombre_bd
    [CHARACTER SET charset]
    [COLLATE collation];

Eliminar Base de Datos

DROP DATABASE nombre_bd;

4.3.2. Definición de Tablas

Crear Tabla

CREATE TABLE nombre_tabla (
    columna1 tipo1 [restricciones],
    columna2 tipo2 [restricciones],
    ...
    [restricciones_tabla]
);

-- Ejemplo completo
CREATE TABLE empleados (
    id_empleado INT PRIMARY KEY AUTO_INCREMENT,
    nombre VARCHAR(50) NOT NULL,
    apellido VARCHAR(50) NOT NULL,
    email VARCHAR(100) UNIQUE,
    fecha_contrato DATE DEFAULT CURRENT_DATE,
    salario DECIMAL(10,2) CHECK (salario >= 0),
    id_departamento INT,
    FOREIGN KEY (id_departamento) 
        REFERENCES departamentos(id_departamento)
);

Modificar Tabla

-- Añadir columna
ALTER TABLE nombre_tabla
ADD columna tipo [restricciones];

-- Modificar columna
ALTER TABLE nombre_tabla
MODIFY columna nuevo_tipo;

-- Eliminar columna
ALTER TABLE nombre_tabla
DROP COLUMN columna;

-- Añadir restricción
ALTER TABLE nombre_tabla
ADD CONSTRAINT nombre_restriccion tipo_restriccion;

Eliminar Tabla

DROP TABLE nombre_tabla;

4.3.3. Definición de Vistas

Crear Vista

CREATE VIEW nombre_vista AS
SELECT columna1, columna2, ...
FROM tabla
WHERE condicion;

-- Ejemplo
CREATE VIEW empleados_IT AS
SELECT nombre, apellido, salario
FROM empleados
WHERE departamento = 'IT';

Modificar Vista

ALTER VIEW nombre_vista AS
SELECT nueva_consulta;

Eliminar Vista

DROP VIEW nombre_vista;

4.3.4. Definición de Índices

Crear Índice

-- Índice simple
CREATE INDEX nombre_indice
ON tabla(columna);

-- Índice único
CREATE UNIQUE INDEX nombre_indice
ON tabla(columna);

-- Índice compuesto
CREATE INDEX nombre_indice
ON tabla(columna1, columna2);

-- Ejemplo
CREATE INDEX idx_apellido
ON empleados(apellido);

Eliminar Índice

DROP INDEX nombre_indice ON tabla;

Consideraciones sobre Índices:

  • Los índices mejoran la velocidad de búsqueda
  • Pero ralentizan las operaciones de inserción y actualización
  • Ocupan espacio adicional en disco
  • Se deben crear en columnas frecuentemente consultadas

4.3.5. Definición de Tipos de Datos

Crear Tipo de Dato (PostgreSQL)

CREATE TYPE estado_civil AS ENUM 
    ('soltero', 'casado', 'divorciado', 'viudo');

CREATE TYPE direccion AS (
    calle VARCHAR(50),
    numero INT,
    ciudad VARCHAR(30),
    cp VARCHAR(5)
);

Usar Tipo Personalizado

CREATE TABLE personas (
    id INT PRIMARY KEY,
    nombre VARCHAR(50),
    estado estado_civil,
    dir direccion
);

Eliminar Tipo

DROP TYPE nombre_tipo;

La creación de tipos personalizados no está disponible en todos los SGBD. MySQL, por ejemplo, no soporta esta característica directamente.

Buenas Prácticas DDL

  • Usar nombres descriptivos y consistentes
  • Documentar la estructura de la base de datos
  • Establecer restricciones apropiadas
  • Mantener la integridad referencial
  • Optimizar el uso de índices
  • Considerar la escalabilidad
