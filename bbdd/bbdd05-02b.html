5.2.1. Control de Acceso a los Datos

1. Gestión de Usuarios

Crear Usuarios

-- Crear usuario con contraseña
CREATE USER 'usuario1'@'localhost' 
IDENTIFIED BY 'password123';

-- Crear usuario que se puede conectar desde cualquier host
CREATE USER 'usuario2'@'%' 
IDENTIFIED BY 'password123';

-- Crear usuario con autenticación por plugin
CREATE USER 'usuario3'@'localhost' 
IDENTIFIED WITH mysql_native_password BY 'password123';

Modificar Usuarios

-- Cambiar contraseña
ALTER USER 'usuario1'@'localhost' 
IDENTIFIED BY 'nueva_password';

-- Cambiar método de autenticación
ALTER USER 'usuario1'@'localhost' 
IDENTIFIED WITH mysql_native_password BY 'password123';

-- Bloquear/Desbloquear cuenta
ALTER USER 'usuario1'@'localhost' ACCOUNT LOCK;
ALTER USER 'usuario1'@'localhost' ACCOUNT UNLOCK;

Eliminar Usuarios

DROP USER 'usuario1'@'localhost';

2. Privilegios

Tipos de Privilegios

Nivel Privilegios Descripción
Global ALL PRIVILEGES Todos los privilegios en todas las bases de datos
Base de datos CREATE, DROP Crear/eliminar objetos en una base de datos
Tabla SELECT, INSERT, UPDATE, DELETE Operaciones sobre datos en tablas
Columna SELECT, INSERT, UPDATE Operaciones sobre columnas específicas

Otorgar Privilegios

-- Privilegios globales
GRANT ALL PRIVILEGES ON *.* 
TO 'admin'@'localhost';

-- Privilegios sobre base de datos
GRANT CREATE, DROP ON empresa.* 
TO 'dba'@'localhost';

-- Privilegios sobre tabla
GRANT SELECT, INSERT ON empresa.empleados 
TO 'rrhh'@'localhost';

-- Privilegios sobre columnas específicas
GRANT SELECT (nombre, departamento), 
UPDATE (salario) ON empresa.empleados 
TO 'supervisor'@'localhost';

-- Privilegios con condiciones
GRANT SELECT ON empresa.empleados 
TO 'analista'@'localhost'
WHERE departamento = 'IT';

Revocar Privilegios

-- Revocar privilegios específicos
REVOKE INSERT, UPDATE ON empresa.empleados 
FROM 'usuario1'@'localhost';

-- Revocar todos los privilegios
REVOKE ALL PRIVILEGES ON *.* 
FROM 'usuario1'@'localhost';

3. Roles

Crear y Gestionar Roles

-- Crear rol
CREATE ROLE 'rol_rrhh';

-- Asignar privilegios al rol
GRANT SELECT, INSERT, UPDATE 
ON empresa.empleados TO 'rol_rrhh';

-- Asignar rol a usuario
GRANT 'rol_rrhh' TO 'usuario1'@'localhost';

-- Activar rol
SET ROLE 'rol_rrhh';

-- Revocar rol
REVOKE 'rol_rrhh' FROM 'usuario1'@'localhost';

4. Vistas de Seguridad

-- Crear vista con acceso restringido
CREATE VIEW empleados_vista AS
SELECT id, nombre, departamento
FROM empleados
WHERE departamento = 'IT';

-- Otorgar acceso a la vista
GRANT SELECT ON empresa.empleados_vista 
TO 'usuario1'@'localhost';

5. Auditoría

-- Habilitar registro de auditoría
SET GLOBAL general_log = 'ON';
SET GLOBAL general_log_file = '/var/log/mysql/mysql.log';

-- Consultar actividad de usuarios
SELECT * FROM mysql.general_log;

-- Ver privilegios actuales
SHOW GRANTS;

-- Ver privilegios de un usuario específico
SHOW GRANTS FOR 'usuario1'@'localhost';

Buenas Prácticas de Seguridad

  • Principio de mínimo privilegio: otorgar solo los permisos necesarios
  • Usar roles para gestionar grupos de privilegios
  • Cambiar contraseñas regularmente
  • Auditar accesos y cambios regularmente
  • Revocar privilegios innecesarios
  • Usar vistas para restringir acceso a datos sensibles
  • Mantener actualizados los sistemas de seguridad

Ejemplo Completo de Configuración de Seguridad

-- Crear roles para diferentes departamentos
CREATE ROLE 'rol_rrhh', 'rol_finanzas', 'rol_it';

-- Configurar privilegios para RRHH
GRANT SELECT, INSERT, UPDATE ON empresa.empleados TO 'rol_rrhh';
GRANT SELECT ON empresa.departamentos TO 'rol_rrhh';

-- Configurar privilegios para Finanzas
GRANT SELECT, UPDATE (salario) ON empresa.empleados TO 'rol_finanzas';
GRANT ALL PRIVILEGES ON empresa.contabilidad TO 'rol_finanzas';

-- Configurar privilegios para IT
GRANT ALL PRIVILEGES ON empresa.* TO 'rol_it';

-- Crear usuarios y asignar roles
CREATE USER 'ana_rrhh'@'localhost' IDENTIFIED BY 'password123';
GRANT 'rol_rrhh' TO 'ana_rrhh'@'localhost';

CREATE USER 'pedro_finanzas'@'localhost' IDENTIFIED BY 'password456';
GRANT 'rol_finanzas' TO 'pedro_finanzas'@'localhost';

-- Crear vista segura para RRHH
CREATE VIEW vista_empleados_rrhh AS
SELECT id, nombre, departamento, fecha_contrato
FROM empleados
WHERE activo = 1;

GRANT SELECT ON empresa.vista_empleados_rrhh TO 'rol_rrhh';
