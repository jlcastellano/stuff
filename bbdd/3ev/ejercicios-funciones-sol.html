Ejercicio 1 - Calificación Numérica a Texto

Crear una función llamada convertir_calificacion que reciba una nota numérica (0-10) y devuelva su equivalente en texto según la siguiente escala:

  • 0-4.99: "Suspenso"
  • 5-6.99: "Aprobado"
  • 7-8.99: "Notable"
  • 9-10: "Sobresaliente"
DELIMITER $$

CREATE FUNCTION convertir_calificacion(nota DECIMAL(4,2))
RETURNS VARCHAR(20)
BEGIN
    DECLARE calificacion VARCHAR(20);
    
    CASE
        WHEN nota < 5 THEN SET calificacion = 'Suspenso';
        WHEN nota < 7 THEN SET calificacion = 'Aprobado';
        WHEN nota < 9 THEN SET calificacion = 'Notable';
        ELSE SET calificacion = 'Sobresaliente';
    END CASE;
    
    RETURN calificacion;
END $$

DELIMITER ;

Ejercicio 2 - Conversor de Medidas

Crear una función llamada convertir_medida que reciba tres parámetros:

  • valor (DECIMAL)
  • unidad_origen (VARCHAR(2)) - 'km', 'm', 'cm'
  • unidad_destino (VARCHAR(2)) - 'km', 'm', 'cm'

La función debe convertir el valor entre las diferentes unidades de medida.

DELIMITER $$

CREATE FUNCTION convertir_medida(
    valor DECIMAL(10,2),
    unidad_origen VARCHAR(2),
    unidad_destino VARCHAR(2)
)
RETURNS DECIMAL(10,2)
BEGIN
    DECLARE resultado DECIMAL(10,2);
    
    -- Primero convertimos todo a metros
    CASE unidad_origen
        WHEN 'km' THEN SET resultado = valor * 1000;
        WHEN 'm' THEN SET resultado = valor;
        WHEN 'cm' THEN SET resultado = valor / 100;
    END CASE;
    
    -- Luego convertimos de metros a la unidad destino
    CASE unidad_destino
        WHEN 'km' THEN SET resultado = resultado / 1000;
        WHEN 'm' THEN SET resultado = resultado;
        WHEN 'cm' THEN SET resultado = resultado * 100;
    END CASE;
    
    RETURN resultado;
END $$

DELIMITER ;

Ejercicio 3 - Calculadora de Edad

Crear una función llamada calcular_edad que reciba una fecha de nacimiento y devuelva la edad actual en años. La función debe manejar fechas inválidas retornando NULL.

DELIMITER $$

CREATE FUNCTION calcular_edad(fecha_nacimiento DATE)
RETURNS INT
BEGIN
    DECLARE edad INT;
    
    IF fecha_nacimiento IS NULL OR fecha_nacimiento > CURDATE() THEN
        RETURN NULL;
    END IF;
    
    SET edad = YEAR(CURDATE()) - YEAR(fecha_nacimiento);
    
    IF DATE_FORMAT(CURDATE(), '%m%d') < DATE_FORMAT(fecha_nacimiento, '%m%d') THEN
        SET edad = edad - 1;
    END IF;
    
    RETURN edad;
END $$

DELIMITER ;

Ejercicio 4 - Generador de Contraseñas

Crear una función llamada generar_password que reciba dos parámetros:

  • longitud (INT) - longitud deseada de la contraseña
  • nivel_seguridad (INT) - 1 (solo letras), 2 (letras y números), 3 (letras, números y símbolos)

La función debe devolver una contraseña aleatoria según los parámetros especificados.

DELIMITER $$

CREATE FUNCTION generar_password(longitud INT, nivel_seguridad INT)
RETURNS VARCHAR(50)
BEGIN
    DECLARE charset VARCHAR(100);
    DECLARE resultado VARCHAR(50);
    DECLARE i INT DEFAULT 1;
    
    CASE nivel_seguridad
        WHEN 1 THEN SET charset = 'abcdefghijklmnopqrstuvwxyzABCDEFGHIJKLMNOPQRSTUVWXYZ';
        WHEN 2 THEN SET charset = 'abcdefghijklmnopqrstuvwxyzABCDEFGHIJKLMNOPQRSTUVWXYZ0123456789';
        ELSE SET charset = 'abcdefghijklmnopqrstuvwxyzABCDEFGHIJKLMNOPQRSTUVWXYZ0123456789!@#$%^&*';
    END CASE;
    
    SET resultado = '';
    
    WHILE i <= longitud DO
        SET resultado = CONCAT(resultado, SUBSTRING(charset, FLOOR(1 + RAND() * LENGTH(charset)), 1));
        SET i = i + 1;
    END WHILE;
    
    RETURN resultado;
END $$

DELIMITER ;

Ejercicio 5 - Formato de Número Telefónico

Crear una función llamada formatear_telefono que reciba un número de teléfono (como cadena) y un código de país, y lo devuelva formateado según el patrón: +[código_país] (XXX) XXX-XXXX

DELIMITER $$

CREATE FUNCTION formatear_telefono(numero VARCHAR(20), codigo_pais INT)
RETURNS VARCHAR(20)
BEGIN
    DECLARE numero_limpio VARCHAR(10);
    
    -- Eliminar caracteres no numéricos
    SET numero_limpio = REGEXP_REPLACE(numero, '[^0-9]', '');
    
    -- Verificar longitud
    IF LENGTH(numero_limpio) != 10 THEN
        RETURN NULL;
    END IF;
    
    RETURN CONCAT(
        '+', codigo_pais, ' ',
        '(', SUBSTRING(numero_limpio, 1, 3), ') ',
        SUBSTRING(numero_limpio, 4, 3), '-',
        SUBSTRING(numero_limpio, 7)
    );
END $$

DELIMITER ;

Ejercicio 6 - Calculadora de IMC con Clasificación

Crear una función llamada calcular_imc_clasificacion que reciba el peso (en kg) y la altura (en metros) de una persona y devuelva una cadena con el IMC calculado y su clasificación según la OMS:

  • < 18.5: "Bajo peso"
  • 18.5-24.9: "Normal"
  • 25-29.9: "Sobrepeso"
  • ≥ 30: "Obesidad"
DELIMITER $$

CREATE FUNCTION calcular_imc_clasificacion(peso DECIMAL(5,2), altura DECIMAL(3,2))
RETURNS VARCHAR(50)
BEGIN
    DECLARE imc DECIMAL(4,2);
    DECLARE clasificacion VARCHAR(20);
    
    IF peso <= 0 OR altura <= 0 THEN
        RETURN NULL;
    END IF;
    
    SET imc = peso / (altura * altura);
    
    CASE
        WHEN imc < 18.5 THEN SET clasificacion = 'Bajo peso';
        WHEN imc < 25 THEN SET clasificacion = 'Normal';
        WHEN imc < 30 THEN SET clasificacion = 'Sobrepeso';
        ELSE SET clasificacion = 'Obesidad';
    END CASE;
    
    RETURN CONCAT('IMC: ', ROUND(imc, 2), ' - ', clasificacion);
END $$

DELIMITER ;
