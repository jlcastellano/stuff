Ejercicios de Manejo de Errores y Transacciones en MySQL

Ejercicio 1 - Sistema de Transferencias

Crear un procedimiento almacenado llamado realizar_transferencia que realice una transferencia entre dos cuentas bancarias. El procedimiento debe:

  • Recibir como parámetros: cuenta_origen, cuenta_destino y monto
  • Verificar que ambas cuentas existan
  • Verificar que haya saldo suficiente
  • Usar transacciones para asegurar la integridad
  • Manejar los posibles errores (cuenta inexistente, saldo insuficiente)

Tabla: cuentas(id, titular, saldo)

DELIMITER $$

CREATE PROCEDURE realizar_transferencia(
    IN p_cuenta_origen INT,
    IN p_cuenta_destino INT,
    IN p_monto DECIMAL(10,2),
    OUT p_mensaje VARCHAR(100)
)
BEGIN
    DECLARE v_saldo_origen DECIMAL(10,2);
    DECLARE EXIT HANDLER FOR SQLEXCEPTION
    BEGIN
        ROLLBACK;
        SET p_mensaje = 'Error en la transacción';
    END;

    -- Verificar cuentas existentes
    IF NOT EXISTS (SELECT 1 FROM cuentas WHERE id = p_cuenta_origen) THEN
        SIGNAL SQLSTATE '45000' SET MESSAGE_TEXT = 'Cuenta origen no existe';
    END IF;

    IF NOT EXISTS (SELECT 1 FROM cuentas WHERE id = p_cuenta_destino) THEN
        SIGNAL SQLSTATE '45000' SET MESSAGE_TEXT = 'Cuenta destino no existe';
    END IF;

    -- Verificar saldo suficiente
    SELECT saldo INTO v_saldo_origen 
    FROM cuentas 
    WHERE id = p_cuenta_origen;

    IF v_saldo_origen < p_monto THEN
        SIGNAL SQLSTATE '45000' SET MESSAGE_TEXT = 'Saldo insuficiente';
    END IF;

    START TRANSACTION;
        -- Realizar transferencia
        UPDATE cuentas 
        SET saldo = saldo - p_monto 
        WHERE id = p_cuenta_origen;

        UPDATE cuentas 
        SET saldo = saldo + p_monto 
        WHERE id = p_cuenta_destino;
    COMMIT;

    SET p_mensaje = 'Transferencia realizada con éxito';
END $$

DELIMITER ;

Ejercicio 2 - Registro de Nuevos Empleados

Crear un procedimiento almacenado llamado registrar_empleado que:

  • Registre un nuevo empleado y su información de contacto
  • Verifique que el DNI no esté duplicado
  • Verifique que el email sea único
  • Maneje los errores usando diferentes tipos de handlers

Tablas: empleados(id, nombre, dni, fecha_ingreso) y contactos(id, empleado_id, email, telefono)

DELIMITER $$

CREATE PROCEDURE registrar_empleado(
    IN p_nombre VARCHAR(100),
    IN p_dni VARCHAR(20),
    IN p_email VARCHAR(100),
    IN p_telefono VARCHAR(20)
)
BEGIN
    DECLARE v_nuevo_id INT;
    DECLARE v_error_msg VARCHAR(100);

    -- Handler para clave duplicada
    DECLARE CONTINUE HANDLER FOR 1062
    BEGIN
        SET v_error_msg = 
            CASE 
                WHEN @error_field = 'dni' THEN 'DNI ya registrado'
                WHEN @error_field = 'email' THEN 'Email ya registrado'
                ELSE 'Valor duplicado'
            END;
        SIGNAL SQLSTATE '45000' SET MESSAGE_TEXT = v_error_msg;
    END;

    START TRANSACTION;
        -- Verificar DNI único
        SET @error_field = 'dni';
        INSERT INTO empleados (nombre, dni, fecha_ingreso)
        VALUES (p_nombre, p_dni, CURDATE());

        SET v_nuevo_id = LAST_INSERT_ID();

        -- Verificar email único
        SET @error_field = 'email';
        INSERT INTO contactos (empleado_id, email, telefono)
        VALUES (v_nuevo_id, p_email, p_telefono);
    COMMIT;

    SELECT 'Empleado registrado correctamente' AS mensaje;
END $$

DELIMITER ;

Ejercicio 3 - Sistema de Inventario

Crear un procedimiento almacenado llamado actualizar_stock que:

  • Actualice el stock de un producto
  • Verifique que el stock no quede negativo
  • Registre el movimiento en un historial
  • Use transacciones y maneje errores específicos

Tablas: productos(id, nombre, stock) y movimientos(id, producto_id, cantidad, tipo, fecha)

DELIMITER $$

CREATE PROCEDURE actualizar_stock(
    IN p_producto_id INT,
    IN p_cantidad INT,
    IN p_tipo ENUM('entrada', 'salida')
)
BEGIN
    DECLARE v_stock_actual INT;
    DECLARE v_stock_nuevo INT;
    
    DECLARE EXIT HANDLER FOR SQLEXCEPTION
    BEGIN
        ROLLBACK;
        SIGNAL SQLSTATE '45000' SET MESSAGE_TEXT = 'Error al actualizar stock';
    END;

    -- Verificar producto existe
    IF NOT EXISTS (SELECT 1 FROM productos WHERE id = p_producto_id) THEN
        SIGNAL SQLSTATE '45000' SET MESSAGE_TEXT = 'Producto no existe';
    END IF;

    SELECT stock INTO v_stock_actual 
    FROM productos 
    WHERE id = p_producto_id;

    -- Calcular nuevo stock
    SET v_stock_nuevo = 
        CASE p_tipo
            WHEN 'entrada' THEN v_stock_actual + p_cantidad
            WHEN 'salida' THEN v_stock_actual - p_cantidad
        END;

    -- Verificar stock suficiente
    IF v_stock_nuevo < 0 THEN
        SIGNAL SQLSTATE '45000' SET MESSAGE_TEXT = 'Stock insuficiente';
    END IF;

    START TRANSACTION;
        -- Actualizar stock
        UPDATE productos 
        SET stock = v_stock_nuevo 
        WHERE id = p_producto_id;

        -- Registrar movimiento
        INSERT INTO movimientos (producto_id, cantidad, tipo, fecha)
        VALUES (p_producto_id, p_cantidad, p_tipo, NOW());
    COMMIT;

    SELECT 'Stock actualizado correctamente' AS mensaje;
END $$

DELIMITER ;

Ejercicio 4 - Proceso de Matrícula

Crear un procedimiento almacenado llamado matricular_estudiante que:

  • Registre la matrícula de un estudiante en un curso
  • Verifique que haya cupo disponible
  • Verifique que el estudiante no esté ya matriculado
  • Maneje múltiples condiciones de error

Tablas: cursos(id, nombre, cupo_maximo, cupo_actual) y matriculas(id, estudiante_id, curso_id, fecha)

DELIMITER $$

CREATE PROCEDURE matricular_estudiante(
    IN p_estudiante_id INT,
    IN p_curso_id INT
)
BEGIN
    DECLARE v_cupo_actual INT;
    DECLARE v_cupo_maximo INT;

    DECLARE EXIT HANDLER FOR SQLWARNING, SQLEXCEPTION
    BEGIN
        ROLLBACK;
        SIGNAL SQLSTATE '45000' SET MESSAGE_TEXT = 'Error en el proceso de matrícula';
    END;

    -- Verificar si ya está matriculado
    IF EXISTS (
        SELECT 1 
        FROM matriculas 
        WHERE estudiante_id = p_estudiante_id 
        AND curso_id = p_curso_id
    ) THEN
        SIGNAL SQLSTATE '45000' 
        SET MESSAGE_TEXT = 'Estudiante ya matriculado en este curso';
    END IF;

    -- Verificar cupo
    SELECT cupo_actual, cupo_maximo 
    INTO v_cupo_actual, v_cupo_maximo
    FROM cursos 
    WHERE id = p_curso_id;

    IF v_cupo_actual >= v_cupo_maximo THEN
        SIGNAL SQLSTATE '45000' 
        SET MESSAGE_TEXT = 'No hay cupo disponible';
    END IF;

    START TRANSACTION;
        -- Registrar matrícula
        INSERT INTO matriculas (estudiante_id, curso_id, fecha)
        VALUES (p_estudiante_id, p_curso_id, NOW());

        -- Actualizar cupo
        UPDATE cursos 
        SET cupo_actual = cupo_actual + 1 
        WHERE id = p_curso_id;
    COMMIT;

    SELECT 'Matrícula realizada con éxito' AS mensaje;
END $$

DELIMITER ;

Ejercicio 5 - Proceso de Facturación

Crear un procedimiento almacenado llamado generar_factura que:

  • Genere una factura con múltiples líneas de detalle
  • Actualice el stock de productos
  • Calcule totales y aplique descuentos
  • Use transacciones y maneje errores de forma granular

Tablas: facturas(id, cliente_id, total, fecha), detalle_factura(id, factura_id, producto_id, cantidad, precio)

DELIMITER $$

CREATE PROCEDURE generar_factura(
    IN p_cliente_id INT,
    IN p_items JSON,
    OUT p_factura_id INT
)
BEGIN
    DECLARE v_total DECIMAL(10,2) DEFAULT 0;
    DECLARE v_item_producto_id INT;
    DECLARE v_item_cantidad INT;
    DECLARE v_item_precio DECIMAL(10,2);
    DECLARE i INT DEFAULT 0;
    
    DECLARE EXIT HANDLER FOR SQLEXCEPTION
    BEGIN
        ROLLBACK;
        SET p_factura_id = NULL;
        SIGNAL SQLSTATE '45000' 
        SET MESSAGE_TEXT = 'Error al generar factura';
    END;

    START TRANSACTION;
        -- Crear factura
        INSERT INTO facturas (cliente_id, total, fecha)
        VALUES (p_cliente_id, 0, NOW());
        
        SET p_factura_id = LAST_INSERT_ID();

        -- Procesar items
        WHILE i < JSON_LENGTH(p_items) DO
            -- Extraer datos del item
            SET v_item_producto_id = JSON_EXTRACT(p_items, CONCAT('$[', i, '].producto_id'));
            SET v_item_cantidad = JSON_EXTRACT(p_items, CONCAT('$[', i, '].cantidad'));
            SET v_item_precio = JSON_EXTRACT(p_items, CONCAT('$[', i, '].precio'));

            -- Verificar stock
            IF NOT EXISTS (
                SELECT 1 
                FROM productos 
                WHERE id = v_item_producto_id 
                AND stock >= v_item_cantidad
            ) THEN
                SIGNAL SQLSTATE '45000' 
                SET MESSAGE_TEXT = 'Stock insuficiente';
            END IF;

            -- Insertar detalle
            INSERT INTO detalle_factura (
                factura_id, 
                producto_id, 
                cantidad, 
                precio
            )
            VALUES (
                p_factura_id,
                v_item_producto_id,
                v_item_cantidad,
                v_item_precio
            );

            -- Actualizar stock
            UPDATE productos 
            SET stock = stock - v_item_cantidad 
            WHERE id = v_item_producto_id;

            -- Actualizar total
            SET v_total = v_total + (v_item_cantidad * v_item_precio);
            SET i = i + 1;
        END WHILE;

        -- Actualizar total en factura
        UPDATE facturas 
        SET total = v_total 
        WHERE id = p_factura_id;
    COMMIT;

    SELECT 'Factura generada correctamente' AS mensaje;
END $$

DELIMITER ;

Ejercicio 6 - Sistema de Reservas

Crear un procedimiento almacenado llamado realizar_reserva que:

  • Gestione una reserva de hotel
  • Verifique disponibilidad de habitaciones
  • Maneje diferentes tipos de errores (fechas inválidas, habitación no disponible)
  • Use transacciones para mantener la integridad

Tablas: habitaciones(id, tipo, estado), reservas(id, habitacion_id, cliente_id, fecha_entrada, fecha_salida)

DELIMITER $$

CREATE PROCEDURE realizar_reserva(
    IN p_habitacion_id INT,
    IN p_cliente_id INT,
    IN p_fecha_entrada DATE,
    IN p_fecha_salida DATE
)
BEGIN
    DECLARE v_disponible BOOLEAN;
    
    DECLARE EXIT HANDLER FOR SQLEXCEPTION
    BEGIN
        ROLLBACK;
        SIGNAL SQLSTATE '45000' 
        SET MESSAGE_TEXT = 'Error al procesar la reserva';
    END;

    -- Validar fechas
    IF p_fecha_entrada >= p_fecha_salida THEN
        SIGNAL SQLSTATE '45000' 
        SET MESSAGE_TEXT = 'Fechas inválidas';
    END IF;

    IF p_fecha_entrada < CURDATE() THEN
        SIGNAL SQLSTATE '45000' 
        SET MESSAGE_TEXT = 'La fecha de entrada debe ser futura';
    END IF;

    -- Verificar disponibilidad
    SELECT NOT EXISTS (
        SELECT 1 
        FROM reservas 
        WHERE habitacion_id = p_habitacion_id
        AND fecha_salida > p_fecha_entrada
        AND fecha_entrada < p_fecha_salida
    ) INTO v_disponible;

    IF NOT v_disponible THEN
        SIGNAL SQLSTATE '45000' 
        SET MESSAGE_TEXT = 'Habitación no disponible para las fechas seleccionadas';
    END IF;

    START TRANSACTION;
        -- Registrar reserva
        INSERT INTO reservas (
            habitacion_id,
            cliente_id,
            fecha_entrada,
            fecha_salida
        ) VALUES (
            p_habitacion_id,
            p_cliente_id,
            p_fecha_entrada,
            p_fecha_salida
        );

        -- Actualizar estado habitación
        UPDATE habitaciones 
        SET estado = 'reservada' 
        WHERE id = p_habitacion_id;
    COMMIT;

    SELECT 'Reserva realizada con éxito' AS mensaje;
END $$

DELIMITER ;
